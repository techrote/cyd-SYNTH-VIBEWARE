cmake_minimum_required(VERSION 3.16)
project(xenobio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(Threads REQUIRED)
add_executable(xenobio_replay replay.cpp)
add_executable(xenobio_runtime_test runtime_test.cpp)
target_link_libraries(xenobio_runtime_test PRIVATE Threads::Threads)
# Assertions exercise real operations; keep them enabled even in Release.
foreach(t xenobio_replay xenobio_runtime_test)
  if(MSVC)
    target_compile_options(${t} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${t} PRIVATE -UNDEBUG)
  endif()
endforeach()
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/media")
add_test(NAME core_replay COMMAND xenobio_replay "${CMAKE_CURRENT_BINARY_DIR}/media")
add_test(NAME runtime_replay COMMAND xenobio_runtime_test "${CMAKE_CURRENT_BINARY_DIR}/runtime.ppm")
set_tests_properties(core_replay runtime_replay PROPERTIES TIMEOUT 90)
if(WIN32)
  add_executable(Xenobio windows.cpp)
  target_link_libraries(Xenobio PRIVATE Threads::Threads winmm gdi32 user32)
endif()
