cmake_minimum_required(VERSION 3.16)
project(chronogrid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(chronogrid_replay replay.cpp)
enable_testing()
add_test(NAME replay COMMAND chronogrid_replay)
if(WIN32)
  add_executable(chronogrid windows.cpp)
  target_link_libraries(chronogrid PRIVATE winmm gdi32 user32)
endif()
