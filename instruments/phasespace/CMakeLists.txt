cmake_minimum_required(VERSION 3.16)
project(phasespace LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(phasespace_replay replay.cpp)
enable_testing()
add_test(NAME replay COMMAND phasespace_replay)
if(WIN32)
  add_executable(phasespace windows.cpp)
  target_link_libraries(phasespace PRIVATE winmm gdi32 user32)
endif()
