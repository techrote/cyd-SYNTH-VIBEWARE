cmake_minimum_required(VERSION 3.16)
project(cyd_cluster LANGUAGES CXX)
add_executable(cluster_native native.cpp)
target_compile_features(cluster_native PRIVATE cxx_std_17)
set_target_properties(cluster_native PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(cluster_native PRIVATE /W4 /WX)
else()
  target_compile_options(cluster_native PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME cluster_native COMMAND cluster_native)

add_executable(cluster_udp udp.cpp)
target_compile_features(cluster_udp PRIVATE cxx_std_17)
set_target_properties(cluster_udp PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(cluster_udp PRIVATE /W4 /WX)
  target_link_libraries(cluster_udp PRIVATE ws2_32)
else()
  target_compile_options(cluster_udp PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME cluster_udp COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/udp_test.py $<TARGET_FILE:cluster_udp>)
set_tests_properties(cluster_udp PROPERTIES TIMEOUT 40)
