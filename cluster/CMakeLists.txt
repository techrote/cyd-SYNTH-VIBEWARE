cmake_minimum_required(VERSION 3.16)
project(cyd_cluster LANGUAGES CXX)
add_executable(cluster_native native.cpp)
target_compile_features(cluster_native PRIVATE cxx_std_17)
set_target_properties(cluster_native PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(cluster_native PRIVATE /W4 /WX)
else()
  target_compile_options(cluster_native PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME cluster_native COMMAND cluster_native)
